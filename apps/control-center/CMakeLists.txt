cmake_minimum_required(VERSION 3.21)
project(astraeus-control-center VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
qt_standard_project_setup(REQUIRES 6.8)
qt_add_executable(astraeus-control-center main.cpp controller.cpp controller.h)
qt_add_qml_module(astraeus-control-center
    URI org.astraeus.control
    QML_FILES Main.qml State.js)
target_link_libraries(astraeus-control-center PRIVATE Qt6::Quick Qt6::QuickControls2)
install(TARGETS astraeus-control-center RUNTIME DESTINATION bin)
install(FILES org.astraeus.ControlCenter.desktop DESTINATION share/applications)
install(FILES org.astraeus.ControlCenter.metainfo.xml DESTINATION share/metainfo)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test QuickTest)
    qt_add_executable(control-center-tests tests/controller_test.cpp controller.cpp controller.h)
    target_include_directories(control-center-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(control-center-tests PRIVATE Qt6::Test Qt6::Gui)
    add_test(NAME control-center-controller COMMAND control-center-tests)
    qt_add_executable(control-center-qml-tests tests/qml_test.cpp)
    target_compile_definitions(control-center-qml-tests PRIVATE QUICK_TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests")
    target_link_libraries(control-center-qml-tests PRIVATE Qt6::QuickTest Qt6::Qml)
    add_test(NAME control-center-qml COMMAND control-center-qml-tests)
    set_tests_properties(control-center-controller control-center-qml PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
endif()
